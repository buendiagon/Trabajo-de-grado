\section*{Glosario}

\glosario[Blosc2]{Biblioteca de compresión por bloques utilizada en este trabajo para el almacenamiento sin pérdida}
\glosario[HPC]{Computación de alto rendimiento (High Performance Computing)}
\glosario[LRU]{Least Recently Used, política de reemplazo de caché basada en el uso más reciente}
\glosario[MPI]{Message Passing Interface, estándar de comunicación para paralelismo en memoria distribuida}
\glosario[MPS]{Matrix Product States, representación tensorial eficiente cuando el entrelazamiento del estado es limitado}
\glosario[OpenMP]{API de paralelismo de memoria compartida para programación multihilo}
\glosario[QFT]{Transformada Cuántica de Fourier (Quantum Fourier Transform)}
\glosario[ZFP]{Biblioteca de compresión para arreglos de punto flotante, empleada aquí como referencia de compresión con pérdida controlada}
